{# Field helpers. Every error is text starting with "Error:", tied to its field by aria-describedby, and listed in the summary at the top of the form. #} {% macro describedby(id, errors, hint=false) -%} {%- set ids = [] -%} {%- if hint %}{% set _ = ids.append(id ~ '-hint') %}{% endif -%} {%- if errors.get(id) %}{% set _ = ids.append(id ~ '-error') %}{% endif -%} {%- if ids %} aria-describedby="{{ ids|join(' ') }}"{% endif -%} {%- if errors.get(id) %} aria-invalid="true"{% endif -%} {%- endmacro %} {% macro error(id, errors) -%} {%- if errors.get(id) %}

Error: {{ errors[id] }}

{% endif -%} {%- endmacro %} {% macro summary(errors, labels={}) -%} {% if errors %}

There {{ 'is a problem' if errors|length == 1 else 'are ' ~ errors|length ~ ' problems' }}

    {% for id, msg in errors.items() %} {% if id == 'form' %}
  • {{ msg }}
    • {% else %}
  • {{ labels.get(id, '') }}{{ ': ' if labels.get(id) }}{{ msg }}
    • {% endif %} {% endfor %}
{% endif %} {%- endmacro %}
